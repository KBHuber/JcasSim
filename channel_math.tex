A Sionna \texttt{PathSolver} solve gives, per path $i$ and per rx/tx antenna pair $(r,s)$, a
passband path coefficient $a_{r,s,i}$ (\texttt{Paths.a}), a path delay $\tau_i$ (\texttt{Paths.tau}),
and a Doppler shift $f_{\Delta,i}$ (\texttt{Paths.doppler}). Both channels consume the same triplet.
$f$ is the carrier frequency, $W$ the sampling frequency, $N$ the frame length, $n$ the sample index.

\begin{center}
\begin{tabular}{@{}l l l@{}}
\toprule
symbol & Sionna term & attribute \\
\midrule
$a_{r,s,i}$ ($a_i$) & passband path coefficient & \texttt{Paths.a} \\
$\tau_i$ & path delay & \texttt{Paths.tau} \\
$f_{\Delta,i}$ & Doppler shift & \texttt{Paths.doppler} \\
$f$ & carrier frequency & \texttt{Scene.frequency} \\
$\ab_i(t)$ & baseband-equivalent channel coefficient & \texttt{cir()} output \\
$\hat h(f,t)$ & channel frequency response & \texttt{Paths.cfr} \\
\bottomrule
\end{tabular}
\end{center}

\section*{1\quad Shared foundation}

\textbf{(1) Doppler shift.} Geometric, from interaction velocities $\mathbf{v}_i$ and ray directions
$\hat{\mathbf{k}}_i$ (subscript here indexes interaction points). Read, not recomputed.
\begin{equation}
f_{\Delta} = \frac{1}{\lambda}\!\left(\mathbf{v}_0^{\!\top}\hat{\mathbf{k}}_0
- \mathbf{v}_{n+1}^{\!\top}\hat{\mathbf{k}}_n
+ \sum_{i=1}^{n}\mathbf{v}_i^{\!\top}\!\big(\hat{\mathbf{k}}_i-\hat{\mathbf{k}}_{i-1}\big)\right),
\qquad \lambda = c/f
\end{equation}
\eqsrc{sionna/rt/path\_solvers/paths.py:379--385}

\textbf{(2) Baseband-equivalent channel coefficient} (Sionna \texttt{cir()}, verbatim). The pivot both
channels build on; static factor $\ab_i$, Doppler phase shift.
\begin{equation}
\ab_i(t) = \underbrace{a_i\,e^{-j2\pi f\tau_i}}_{\ab_i}\;
\underbrace{e^{\,j2\pi f_{\Delta,i}t}}_{\text{Doppler phase shift}}
\end{equation}
\eqsrc{sionna/rt/path\_solvers/paths.py:404--405, 494--520}

\textbf{(3) Channel frequency response} (Sionna \texttt{cfr()}; here $f$ is the frequency variable),
and the equivalent time-domain echo both channels realize.
\begin{equation}
\hat h(f,t) = \sum_{i=0}^{M-1}\ab_i(t)\,e^{-j2\pi f\tau_i}
\qquad\Longleftrightarrow\qquad
y_r^{(i)}(t) = \sum_s \ab_{r,s,i}\,e^{\,j2\pi f_{\Delta,i}t}\,x_s(t-\tau_i)
\end{equation}

\section*{2\quad Stock \texttt{SionnaRTChannel}}

\textbf{(4) Densified CIR.} \texttt{cir()} materializes $\ab$ on the transmit grid $t_n=n/W$ for every
path and sample $\Rightarrow$ tensor $[N_{\text{rx}},N_{\text{tx}},\text{paths},N]$ (the OOM).
\begin{equation}
a_{r,s,i}[n] = \ab_{r,s,i}(t_n) = a_{r,s,i}\,e^{-j2\pi f\tau_i}\,e^{\,j2\pi f_{\Delta,i}t_n},
\qquad t_n = n/W,\; n=0,\dots,N-1
\end{equation}
\eqsrc{hermespy/channel/sionna\_rt\_channel.py:142--150}

\textbf{(5) Delay tap.} Snap to whole samples.
\begin{equation}
\ell_i = \flr{\tau_i W}\in\mathbb{Z}_{\ge 0}
\end{equation}
\eqsrc{hermespy/channel/sionna\_rt\_channel.py:177}

\textbf{(6) Propagate.} Shift each path by $\ell_i$, scale, sum; $\sqrt{g}$ once. Doppler lands at
$(n-\ell_i)/W$.
\begin{equation}
y_r[n] = \sqrt{g}\sum_i\sum_s \ab_{r,s,i}\!\Big(\tfrac{n-\ell_i}{W}\Big)\,x_s[n-\ell_i],
\qquad 0\le n-\ell_i\le N-1
\end{equation}
\eqsrc{hermespy/channel/sionna\_rt\_channel.py:174--182}

\textbf{(7) State (CSI).} Same physics as a tap tensor; each path into tap $\ell=\flr{\tau_i W}$,
Doppler at $n/W$.
\begin{equation}
H_{r,s}[n,\ell] = \sqrt{g}\!\!\sum_{\flr{\tau_i W}=\ell}\!\! \ab_{r,s,i}(n/W),
\qquad y_r[n]=\sum_{s,\ell}H_{r,s}[n,\ell]\,x_s[n-\ell]
\end{equation}
\eqsrc{hermespy/channel/sionna\_rt\_channel.py:120--129}

\section*{3\quad Geometric \texttt{GeometricChannelResponse}}

\textbf{(8) Carrier fold.} Raw \texttt{paths.a} omits the carrier delay phase $e^{-j2\pi f\tau_i}$
(Sionna adds it only inside \texttt{cir()}); this puts it back, giving $\ab_i$ of (2). No time axis stored.
\begin{equation}
\ab_{r,s,i} = a_{r,s,i}\,e^{-j2\pi f\tau_i}
\end{equation}
\eqsrc{geometric\_sionna\_channel.py:155}

\textbf{(9) Delay} $\equiv$ stock (5)+(6). Same floor, then a causal shift: zeros in front, tail dropped
--- the stock tap into a zeroed buffer, computed directly.
\begin{equation}
\ell_i = \flr{\tau_i W},\qquad
\tilde x_{s,i}[n] = \begin{cases} x_s[n-\ell_i], & n\ge \ell_i\\[2pt] 0, & n<\ell_i \end{cases}
\end{equation}
\eqsrc{geometric\_sionna\_channel.py:233--234 (\texttt{\_causal\_delay})}

\textbf{(10) Propagate.} Delay, multiply by the closed-form Doppler at absolute time $n/W+t_0$, scale,
sum over kept paths (delay past the self-coupling gate $\tau_i\ge 2r_{\min}/c$; no strength cap).
\begin{equation}
y_r[n] = \sqrt{P_w}\sum_i\sum_s \ab_{r,s,i}\,\tilde x_{s,i}[n]\,e^{\,j2\pi f_{\Delta,i}(n/W+t_0)}
\end{equation}
\eqsrc{geometric\_sionna\_channel.py:223, 232--236, 250}

\textbf{(11) State (CSI).} Mirrors stock (7).
\begin{equation}
H_{r,s}[n,\ell] = \sqrt{g}\!\!\sum_{\lfloor\tau_i W\rfloor=\ell}\!\! \ab_{r,s,i}\,e^{\,j2\pi f_{\Delta,i}n/W}
\end{equation}
\eqsrc{geometric\_sionna\_channel.py:324--333}

\section*{5\quad When $y_r^{\text{stock}}[n]=y_r^{\text{geo}}[n]$}

\begin{enumerate}\setlength{\itemsep}{1pt}
\item Same solve: matching arrays, \texttt{diffuse\_reflection}, seed, depth, so the path sets
$\{(a_{r,s,i},\tau_i,f_{\Delta,i})\}$ match (override the stock $1\times1$ iso array and
\texttt{diffuse\_reflection=False}).
\item Same path set: $r_{\min}=0$, both keep every valid path.
\item Delay: automatic --- same floor $\ell_i=\flr{\tau_i W}$ and same causal shift (5,6\,$=$\,9), identical sample-for-sample.
\item Doppler origin aligned: $t_0=0$ and matched reference (use $n/W$, or absorb the fixed per-path
offset $e^{j2\pi f_{\Delta,i}\ell_i/W}$ vs.\ \texttt{\_propagate}).
\item Scaling: $\sqrt{P_w}\sqrt{g_{\text{geo}}}=\sqrt{g}$.
\end{enumerate}

Under 1--5 the outputs are algebraically identical. The delay math is the same; what differs is
memory strategy and the deliberate modelling choices (real array, diffuse on, all paths, self-coupling
gate, absolute Doppler origin). The Doppler-origin difference is at most a constant per-path phase.
